// En server/routes.ts - Endpoint /api/chat
app.post('/api/chat', async (req, res) => {
  const { message, imageId, model = 'gemini' } = req.body;
  
  // Obtener la imagen de la base de datos
  let imageBase64: string | undefined;
  if (imageId) {
    const image = await storage.getImage(imageId);
    if (image && image.url) {
      // Leer el archivo de imagen del servidor
      const response = await fetch(`http://localhost:5000${image.url}`);
      const buffer = await response.buffer();
      
      // CONVERSIÓN A BASE64 - Este es el formato que Gemini necesita
      imageBase64 = buffer.toString('base64');
    }
  }
  
  // Enviar a Gemini con la imagen en base64
  const response = await getChatResponseGemini(messages, imageBase64);
});
Paso 3: Formato de Envío a Gemini API
// En server/services/gemini.ts
export async function getChatResponseGemini(
  messages: Array,
  imageBase64?: string  // La imagen en formato base64
) {
  const ai = new GoogleGenAI({ apiKey: GEMINI_API_KEY });
  
  // ESTRUCTURA DEL MENSAJE CON IMAGEN
  const chatMessages = [
    {
      role: "user",
      parts: [
        {
          // Instrucciones para el análisis
          text: "Analiza esta imagen y describe lo que ves..."
        },
        {
          // LA IMAGEN SE ENVÍA ASÍ:
          inlineData: {
            data: imageBase64,        // String base64 de la imagen
            mimeType: "image/jpeg"    // Tipo MIME de la imagen
          }
        },
        {
          // Mensaje adicional del usuario (opcional)
          text: userMessage
        }
      ]
    }
  ];
  
  // Llamada a la API de Gemini
  const response = await ai.models.generateContent({
    model: "gemini-2.0-flash-exp",
    contents: chatMessages
  });
  
  return response.text;
}
2. Formato Específico de la Imagen para Gemini
// El formato que Gemini espera es:
{
  inlineData: {
    data: "iVBORw0KGgoAAAANS...",  // String base64 (sin el prefijo data:image/jpeg;base64,)
    mimeType: "image/jpeg"         // Puede ser: image/jpeg, image/png, image/gif, etc.
  }
}
3. Ejemplo Completo de Implementación Simplificada
// Función para convertir imagen a base64
function imageToBase64(imagePath) {
  const fs = require('fs');
  const imageBuffer = fs.readFileSync(imagePath);
  return imageBuffer.toString('base64');
}
// Función para enviar imagen a Gemini
async function analyzeImageWithGemini(imagePath, userPrompt) {
  const { GoogleGenAI } = require("@google/genai");
  
  // 1. Convertir imagen a base64
  const imageBase64 = imageToBase64(imagePath);
  
  // 2. Configurar cliente de Gemini
  const genAI = new GoogleGenAI({ 
    apiKey: "tu_api_key_aqui" 
  });
  
  // 3. Preparar mensaje con imagen
  const message = {
    role: "user",
    parts: [
      {
        text: userPrompt || "Describe esta imagen en detalle"
      },
      {
        inlineData: {
          data: imageBase64,
          mimeType: "image/jpeg"
        }
      }
    ]
  };
  
  // 4. Enviar a Gemini
  const result = await genAI.models.generateContent({
    model: "gemini-2.0-flash-exp",
    contents: [message]
  });
  
  // 5. Obtener respuesta
  return result.text;
}

4. Ejemplo con Node.js y Express
const express = require('express');
const multer = require('multer');
const { GoogleGenAI } = require("@google/genai");
const app = express();
const upload = multer({ dest: 'uploads/' });
app.post('/analyze-image', upload.single('image'), async (req, res) => {
  try {
    // 1. Leer archivo subido
    const fs = require('fs');
    const imageBuffer = fs.readFileSync(req.file.path);
    const imageBase64 = imageBuffer.toString('base64');
    
    // 2. Configurar Gemini
    const genAI = new GoogleGenAI({ 
      apiKey: process.env.GEMINI_API_KEY 
    });
    
    // 3. Enviar a Gemini
    const result = await genAI.models.generateContent({
      model: "gemini-2.0-flash-exp",
      contents: [{
        role: "user",
        parts: [
          { text: req.body.prompt || "Describe esta imagen" },
          { 
            inlineData: {
              data: imageBase64,
              mimeType: req.file.mimetype
            }
          }
        ]
      }]
    });
    
    // 4. Retornar respuesta
    res.json({ 
      description: result.text 
    });
    
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});
Puntos Clave:
La imagen DEBE estar en formato base64 (no URL)
No incluyas el prefijo data:image/jpeg;base64, en el string base64
Especifica el mimeType correcto (image/jpeg, image/png, etc.)
El modelo usado es gemini-2.0-flash-exp
La imagen se envía dentro de inlineData en el array de parts
